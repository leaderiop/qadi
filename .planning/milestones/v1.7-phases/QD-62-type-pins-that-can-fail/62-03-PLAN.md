### `<S>/SweepReporter.mjs`

```js
// The flip sweep's reporter: one line per project and per assertion, with the raw
// diagnostic codes tstyche's own text output drops. Scratch; never committed.
export default class SweepReporter {
  on([event, payload]) {
    if (event === "project:uses") {
      console.log("SWEEP-PROJECT " + payload.projectConfig.specifier);
      return;
    }
    if (!event.startsWith("expect:") || event === "expect:start") return;
    const node = payload.result.expect;
    const sf = node.matcherNode.getSourceFile();
    const { line } = sf.getLineAndCharacterOfPosition(node.matcherNameNode.name.getStart());
    const diags = [...(node.abilityDiagnostics ?? [])].map((d) => ({ code: d.code, source: d.source ?? "ts" }));
    console.log(
      "SWEEP " +
        JSON.stringify({ file: sf.fileName, line: line + 1, matcher: node.matcherNameNode.name.text, not: node.isNot, event: event.slice(7), diags }),
    );
  }
}
```

### `<S>/sweep.mjs`

```js
// The flip sweep: copy each type test holding a negative pin to `zz-flip-<name>` beside it,
// flip every negative pin to its positive form, run tstyche (tstyche.json's config) with
// SweepReporter, delete the copies, and require every flipped pin to fail with an allowed
// TypeScript code and nothing else. Scratch; never committed. From the repository root:
//   node <scratch>/sweep.mjs <scratch>/SweepReporter.mjs [file ...]
import cp from "node:child_process";
import fs from "node:fs";
import path from "node:path";
import { pathToFileURL } from "node:url";

const [reporter, ...only] = process.argv.slice(2);
const ALLOWED = new Set([2322, 2345, 2353, 2375, 2379, 2554, 2559, 2769]);
const EXCLUDED = "packages/core/test/FloatingEffectPins.tst.ts";
const NEG = /\.\s*not\s*\.\s*(toBeCallableWith|toBeConstructableWith)\b/;
const NEG_ALL = /\.\s*not\s*\.\s*(toBeCallableWith|toBeConstructableWith)\b/g;
const COMMENT = /^\s*(\/\/|\*|\/\*)/;
const PREFIX = "zz-flip-";
const tracked = cp.execFileSync("git", ["ls-files", "*.tst.ts"], { encoding: "utf8" }).split("\n").filter(Boolean);
const files = (only.length > 0 ? only : tracked).filter((f) => f !== EXCLUDED);
const pins = [];
const copies = [];
let out = "";
try {
  for (const f of files) {
    const lines = fs.readFileSync(f, "utf8").split("\n");
    const hits = [];
    lines.forEach((l, i) => {
      if (COMMENT.test(l) || !NEG.test(l)) return;
      const s = /expect\s*(?:<(.+?)>)?\((.*?)\)\s*\.\s*type/.exec(l);
      hits.push({ file: f, line: i + 1, form: NEG.exec(l)[1], subject: s === null ? "?" : (s[1] ?? s[2]) });
    });
    if (hits.length === 0) continue;
    pins.push(...hits);
    const copy = path.join(path.dirname(f), PREFIX + path.basename(f));
    fs.writeFileSync(copy, lines.map((l) => (COMMENT.test(l) ? l : l.replace(NEG_ALL, ".$1"))).join("\n"));
    copies.push(copy);
  }
  const r = cp.spawnSync(
    process.execPath,
    ["node_modules/tstyche/dist/bin.js", "--reporters", pathToFileURL(path.resolve(reporter)).href, PREFIX],
    { encoding: "utf8", env: { ...process.env, NO_COLOR: "1" }, stdio: ["ignore", "pipe", "pipe"], maxBuffer: 1 << 26 },
  );
  out = r.stdout + r.stderr;
} finally {
  for (const c of copies) fs.rmSync(c, { force: true });
}
const projects = [...out.matchAll(/^SWEEP-PROJECT (.*)$/gm)].map((m) => m[1]);
const rows = [...out.matchAll(/^SWEEP (\{.*\})$/gm)].map((m) => JSON.parse(m[1]));
const bad = [];
if (projects.length === 0 || projects.some((p) => !p.endsWith("tsconfig.tstyche.json"))) {
  bad.push("tstyche did not compile the copies under tsconfig.tstyche.json: " + JSON.stringify(projects));
}
const table = pins.map((p, i) => {
  const copyRel = path.join(path.dirname(p.file), PREFIX + path.basename(p.file));
  const row = rows.find((x) => path.relative(process.cwd(), x.file) === copyRel && x.line === p.line && x.not === false && x.matcher === p.form);
  const codes = row === undefined ? [] : row.diags.map((d) => d.code);
  const ok = row !== undefined && row.event === "fail" && codes.length > 0 && row.diags.every((d) => d.source === "ts" && ALLOWED.has(d.code));
  if (!ok) bad.push(p.file + ":" + p.line + " " + JSON.stringify(row ?? "no row"));
  const form = p.form === "toBeCallableWith" ? "callable" : "constructable";
  return "| S" + String(i + 1).padStart(2, "0") + " | `" + p.file + ":" + p.line + "` | `" + p.subject + "` | " + form + " | " + (row === undefined ? "missing" : row.event) + " | " + codes.map((c) => "TS" + c).join(", ") + " |";
});
console.log(table.join("\n"));
console.log("sweep: " + pins.length + " flipped pin(s) in " + copies.length + " file(s), " + bad.length + " outside the oracle");
if (bad.length > 0) {
  console.error(bad.join("\n"));
  process.exit(1);
}
```

### `<S>/overload-inventory.mjs`

```js
// The subjects a negative pin reads through a parameter type: every `Params<…>`,
// `Parameters<…>` or `ConstructorParameters<…>` in a refusal pin, directly or through a
// `type` alias declared in the same file (alias chains followed). Scratch; never committed.
// From the repository root:  node <S>/overload-inventory.mjs
import cp from "node:child_process";
import fs from "node:fs";

const EXCLUDED = "packages/core/test/FloatingEffectPins.tst.ts";
const REFUSAL = /\.\s*not\s*\.\s*(?:toBeCallableWith|toBeConstructableWith|toBeAssignableTo|toBeAssignableFrom)\b/g;
const READ = /\b(?:Params|Parameters|ConstructorParameters)<\s*(?:typeof\s+)?([A-Za-z_$][\w$]*(?:\.[A-Za-z_$][\w$]*)*)\s*(?:\[\s*"([\w$]+)"\s*\])?/g;
const ALIAS = /\btype\s+([A-Za-z_$][\w$]*)(?:<[^=]*?>)?\s*=\s*([^;]+);/g;
const decomment = (s) => s.replace(/\/\*[\s\S]*?\*\//g, (m) => m.replace(/[^\n]/g, " ")).replace(/\/\/[^\n]*/g, (m) => " ".repeat(m.length));
const files = cp.execFileSync("git", ["ls-files", "*.tst.ts"], { encoding: "utf8" }).split("\n").filter((f) => f !== "" && f !== EXCLUDED);
/** @type {Map<string, Set<string>>} */
const subjects = new Map();
for (const file of files) {
  const code = decomment(fs.readFileSync(file, "utf8"));
  const aliases = new Map();
  for (const m of code.matchAll(ALIAS)) aliases.set(m[1], [...(aliases.get(m[1]) ?? []), m[2]]);
  const resolve = (text, seen) => {
    const out = new Set();
    for (const m of text.matchAll(READ)) if (m[2] !== undefined || !aliases.has(m[1])) out.add(m[2] ?? m[1].split(".").pop());
    for (const [name, defs] of aliases) {
      if (seen.has(name) || !new RegExp(`\\b${name.replace(/\$/g, "\\$")}\\b`).test(text)) continue;
      for (const def of defs) for (const s of resolve(def, new Set([...seen, name]))) out.add(s);
    }
    return out;
  };
  for (const m of code.matchAll(REFUSAL)) {
    const start = code.lastIndexOf("expect", m.index);
    const end = code.indexOf(";", m.index);
    const statement = code.slice(start, end === -1 ? code.length : end);
    const line = code.slice(0, m.index).split("\n").length;
    for (const s of resolve(statement, new Set())) {
      if (!subjects.has(s)) subjects.set(s, new Set());
      subjects.get(s).add(`${file}:${line}`);
    }
  }
}
for (const [s, at] of [...subjects].sort(([a], [b]) => a.localeCompare(b))) console.log(`OVERLOAD-SUBJECT ${s} ${[...at].join(" ")}`);
console.log(`overload inventory: ${subjects.size} subject(s)`);
```
