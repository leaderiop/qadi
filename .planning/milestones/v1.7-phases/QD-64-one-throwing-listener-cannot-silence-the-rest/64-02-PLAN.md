  "notifies once when set to a different reference",
  "publish notifies even when the reference is unchanged",
  "stops notifying a listener once it unsubscribes, and only that one",
];

const committed = readFileSync(SEAM, "utf8");
const once = (from, to, times = 1) => (text) => {
  const n = text.split(from).length - 1;
  if (n !== times) throw new Error(`anchor ${JSON.stringify(from)} occurs ${n} time(s), wanted ${times}`);
  return text.split(from).join(to);
};
const VARIANTS = [

  ["v1", "iterate a snapshot of the set", once("for (const listener of listeners) {", "for (const listener of [...listeners]) {"), [C2, C3]],

  ["v5", "count in the message", once(`"More than one of the store's listeners threw."`, "`${thrown.length} of the store's listeners threw.`"), [R2]],
  ["v6", "no early return when nothing threw", once("    if (thrown.length === 0) return;\n", ""), [C1, C2, C3, ...NOTIFIES]],
  ["v7", "swallow what a listener throws", once("thrown.push(error);", "void error;"), [R1, R2, R3, R4, R5, C4]],

}
console.log("| Variant | Edit | Tests | Failed | Result |\n| --- | --- | --- | --- | --- |");
