````js
// Applies one records edit list from the repository root: each edit replaces `old` with `new` where `old` occurs
// exactly once, appends `append`, or inserts `text` after the one line that starts with `after_line_starting`. An edit
// that does not find its text exactly once stops the run before any file is written. Scratch, never committed:
//   node <S>/records-edit.mjs <S>/texts/records-t1.json
import { readFileSync, writeFileSync } from "node:fs";

const [list] = process.argv.slice(2);
const edits = JSON.parse(readFileSync(list, "utf8"));
const files = new Map();
const stop = (message) => {
  console.error(message);
  process.exit(1);
};
for (const e of edits) {
  let s = files.get(e.file) ?? readFileSync(e.file, "utf8");
  if ("append" in e) s += e.append;
  else if ("after_line_starting" in e) {
    const lines = s.split("\n");
    const at = lines.flatMap((line, i) => (line.startsWith(e.after_line_starting) ? [i] : []));
    if (at.length !== 1) stop(`${e.file}: ${at.length} lines start with ${e.after_line_starting}`);
    lines.splice(at[0] + 1, 0, e.text.replace(/\n$/, ""));
    s = lines.join("\n");
  } else {
    const n = s.split(e.old).length - 1;
    if (n !== 1) stop(`${e.file}: the old text occurs ${n} times: ${e.old.slice(0, 80)}`);
    s = s.replace(e.old, () => e.new);
  }
  files.set(e.file, s);
}
for (const [file, s] of files) writeFileSync(file, s);
console.log(`${list}: ${edits.length} edit(s) applied to ${files.size} file(s)`);
````

// node <S>/ccr-row.mjs <phase_base>, from the repository root: every spec document this phase revised is cited in the
// CCR-QD-457 row at its current Revision. Prints `<path> Rev <revision>` for each, then a count; exits 1 naming any
// document the row does not cite at its current Revision. Scratch only, never committed.
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";

const [base] = process.argv.slice(2);
const changed = execFileSync("git", ["diff", "--name-only", base, "--", "spec"], { encoding: "utf8" })
  .split("\n")
  .filter((file) => file.endsWith(".md"));
const rows = readFileSync("spec/README.md", "utf8")
  .split("\n")
  .filter((line) => line.startsWith("| CCR-QD-457 |"));
const escape = (text) => text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
const bad = rows.length === 1 ? [] : [`spec/README.md holds ${rows.length} CCR-QD-457 rows, not one`];
if (changed.length === 0) bad.push("no spec document differs from the phase base");
for (const file of changed) {
  const revision = /^> \| Revision\s*\| ([0-9.]+)/m.exec(readFileSync(file, "utf8"))?.[1];
  if (revision === undefined) {
    bad.push(`${file} has no Revision row`);
    continue;

  const cited = new RegExp(`\`${escape(file)}\` Rev ${escape(revision)}[ ,]`);
