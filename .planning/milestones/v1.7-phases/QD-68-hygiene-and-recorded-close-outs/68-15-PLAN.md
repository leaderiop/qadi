````js
// node <S>/oi-map-check.mjs <summary> <phase_base>, from the repository root. Fails, naming each reason, unless the
// summary's rows `| OI-NN | 68-PP | fix or record | closed by |` hold OI-01 to OI-29 once each, with the plan and
// disposition of the phase's map, and every closed-by cell names, in backticks, the short hash of a commit after
// <phase_base> whose subject carries `(68-PP)`; a record's cell also names CCR-QD-459, whose row in spec/README.md says
// `OI-26 to OI-29 recorded`. OI-18's commit is the first after <phase_base>, and every record's commit comes after every
// fix's. Prints `oi-map: 25 fixed, 4 recorded`. Scratch, never committed.
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";

const [summary, base] = process.argv.slice(2);
if (summary === undefined || base === undefined) {
  console.error("usage: oi-map-check.mjs <summary> <phase_base>");
  process.exit(2);
}
const PLAN = {
  "01": "07", "02": "04", "03": "04", "04": "04", "05": "07", "06": "07", "07": "07", "08": "06", "09": "04", "10": "05",
  "11": "03", "12": "03", "13": "03", "14": "06", "15": "08", "16": "08", "17": "08", "18": "01", "19": "08", "20": "08",
  "21": "08", "22": "09", "23": "09", "24": "01", "25": "09", "26": "11", "27": "11", "28": "11", "29": "11",
};
/** The phase's commits, oldest first. */
const ours = execFileSync("git", ["log", "--reverse", "--format=%H %s", `${base}..HEAD`], { encoding: "utf8" })
  .split("\n")
  .filter(Boolean)
  .map((line) => ({ hash: line.slice(0, 40), subject: line.slice(41) }));
const ccr = readFileSync("spec/README.md", "utf8").split("\n").find((line) => line.startsWith("| CCR-QD-459 |")) ?? "";
const bad = [];
const seen = new Map();
const at = { fix: [], record: [] };
let first;
for (const line of readFileSync(summary, "utf8").split("\n")) {
  const m = /^\| OI-(\d\d) \| 68-(\d\d) \| (fix|record) \| (.*) \|$/.exec(line);
  if (m === null) continue;
  const [, oi, plan, disposition, closedBy] = m;
  seen.set(oi, (seen.get(oi) ?? 0) + 1);
  if (PLAN[oi] !== plan) bad.push(`OI-${oi}: plan 68-${plan}, wanted 68-${PLAN[oi]}`);
  const want = Number(oi) >= 26 ? "record" : "fix";
  if (disposition !== want) bad.push(`OI-${oi}: ${disposition}, wanted ${want}`);
  const named = [...closedBy.matchAll(/`([0-9a-f]{8,40})`/g)].map((h) => h[1]);
  const index = ours.findIndex((c) => named.some((h) => c.hash.startsWith(h)) && c.subject.includes(`(68-${plan})`));
  if (index === -1) {
    bad.push(`OI-${oi}: no commit of 68-${plan} after the phase base is named`);
    continue;
  }
  at[want].push(index);
  if (oi === "18") first = index;
  if (want === "record" && (!closedBy.includes("CCR-QD-459") || !ccr.includes("OI-26 to OI-29 recorded"))) {
    bad.push(`OI-${oi}: not recorded in CCR-QD-459`);
  }
}
for (const oi of Object.keys(PLAN)) {
  if (seen.get(oi) !== 1) bad.push(`OI-${oi}: ${seen.get(oi) ?? 0} row(s), wanted 1`);
}
if (first !== undefined && first !== 0) bad.push("OI-18's fix is not the phase's first commit");
if (at.record.length > 0 && at.fix.length > 0 && Math.min(...at.record) < Math.max(...at.fix)) {
  bad.push("a record's commit comes before a fix's");
}
if (bad.length > 0) {
  console.error(bad.join("\n"));
  process.exit(1);
}
console.log("oi-map: 25 fixed, 4 recorded");
````
