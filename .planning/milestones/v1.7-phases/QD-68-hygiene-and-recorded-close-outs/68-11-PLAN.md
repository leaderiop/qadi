// node <S>/ccr-row.mjs <phase_base>, from the repository root: every spec document this phase revised is cited in the
// CCR-QD-459 row at its current Revision. Prints `<path> Rev <revision>` for each, then a count; exits 1 naming any

import { readFileSync } from "node:fs";

  .split("\n")
  .filter((file) => file.endsWith(".md"));

  .split("\n")
  .filter((line) => line.startsWith("| CCR-QD-459 |"));

    bad.push(`${file} has no Revision row`);

Scratch, `<S>/rethrow-gate.mjs` (SHA-256 `ea216115e04deeb816c5680169c0e2b1d31e0c36b49a8dfad602bbe51755f82a`): the check that no text states the capacity change's
rethrow without its residue.

````js
// node <S>/rethrow-gate.mjs <file>..., from the repository root. Each file states the capacity change's rethrow rule:
// with its line breaks and the comment or quote markers that open a line folded to one space, it holds the phrase
// `rethrown once the read` at least once, and every occurrence is followed, within 400 characters, by the residue
// clause `replaces the timeline listener's` (a status listener that throws on the `Reading` the read publishes over a
// `Stopped` escapes in its place, and no read is started). Fails, naming each file and the text it read, when a file
// states the rethrow unconditionally or not at all. Prints `rethrow-gate: <n> statement(s) in <m> file(s) carry the
// replacement`. Scratch, never committed.
import { readFileSync } from "node:fs";

const files = process.argv.slice(2);
if (files.length === 0) {
  console.error("usage: rethrow-gate.mjs <file>...");
  process.exit(2);
}
const RULE = "rethrown once the read";
const RESIDUE = "replaces the timeline listener's";
const bad = [];
let statements = 0;
for (const file of files) {
  const text = readFileSync(file, "utf8").replace(/[ \t]*\n[ \t]*(?:\*(?!\*)|\/\/|>)?[ \t]*/g, " ");
  let at = text.indexOf(RULE);
  if (at === -1) bad.push(`${file}: does not state the rule (no "${RULE}")`);
  while (at !== -1) {
    statements++;
    if (!text.slice(at, at + 400).includes(RESIDUE)) {
      bad.push(`${file}: states the rethrow unconditionally: "${text.slice(Math.max(0, at - 60), at + 160)}…"`);
    }
    at = text.indexOf(RULE, at + RULE.length);
  }
}
if (bad.length > 0) {
  console.error(bad.join("\n"));
  process.exit(1);
}
console.log(`rethrow-gate: ${statements} statement(s) in ${files.length} file(s) carry the replacement`);
````
