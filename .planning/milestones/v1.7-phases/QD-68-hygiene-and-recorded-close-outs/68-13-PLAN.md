// node <D>68-reverify.mjs <phase> <phase_base> [<heading>], from the repository root, once the changes a re-verification
// answers are committed and before that re-verification of Phase <phase> is. Prints the mechanical head of the section:
// `## <heading>` (default `Re-verification (2026-11-21, Phase 68 sweep)`; a later re-verification names its own date and
// cause), the line `68-stale.mjs` prints for the phase, each changed cited file with the commits since the verification
// that changed it (those before <phase_base> marked), the Co-Authored-By trailers on the phase's own commits, its
// VALIDATION.md status, and the covered_digest line before and after, recomputed over the same covered_files by GSD's
// `verification.fingerprint`. Its last line is the new `covered_digest:` line, written as the file writes it. Extracted
// to <S>/reverify.mjs for the sweep and committed by 68-13 as <D>68-reverify.mjs for any later re-verification; a phase
// record, not run by `pnpm check:pr`.
import { execFileSync } from "node:child_process";
import { existsSync, readdirSync, readFileSync } from "node:fs";

const [phase, base, heading = "Re-verification (2026-11-21, Phase 68 sweep)"] = process.argv.slice(2);
if (!/^6[2-7]$/.test(phase ?? "") || base === undefined) {
  console.error("usage: 68-reverify.mjs <62..67> <phase_base> [<heading>]");
  process.exit(2);
}
const git = (...args) => execFileSync("git", args, { encoding: "utf8", maxBuffer: 1 << 26 });
const cell = (text) => text.replace(/\|/g, "\\|");
const dir = `.planning/phases/${readdirSync(".planning/phases").find((d) => d.startsWith(`QD-${phase}-`))}`;
const doc = `${dir}/${phase}-VERIFICATION.md`;

  encoding: "utf8",

if (at === -1) throw new Error(`68-stale.mjs printed no line for ${phase}`);
const listed = [];
for (let i = at + 1; i < stale.length && stale[i].startsWith("  "); i++) listed.push(stale[i].trim());

  const path = entry.replace(/ \(covered\)$/, "");
  const by = git("log", "--format=%H %s", `${vcommit}..HEAD`, "--", path)
    .split("\n")
    .filter(Boolean)
    .map((line) => `\`${line.slice(0, 8)}\` ${before.has(line.slice(0, 40)) ? "(before Phase 68) " : ""}${line.slice(41)}`);
  return `| \`${path}\`${path === entry ? "" : " (covered)"} | ${cell(by.join("<br>"))} |`;

const counted = [...trailers].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0])).map(([k, n]) => `${n} ${k}`);

````js
// node <S>/sweep-check.mjs <phase> <phase_base> <reverify-output>, from the repository root, once Phase <phase>'s
// re-verification is written. Fails, naming each reason, unless its VERIFICATION.md is the report at <phase_base> kept
// whole with only its covered_digest line replaced by the one reverify.mjs computed, followed by one appended
// `## Re-verification (2026-11-21, Phase 68 sweep)` section that holds reverify.mjs's output verbatim, a
// `### Spot-checks re-run (Phase 68 sweep)` table with at least one row per row of the report's spot-check table (its
// Observable Truths table where it has none), for Phase 62 a `### Truths re-checked (Phase 68 sweep)` table with at
// least one row per Observable Truth, and the line `**Verdict:** passed`. Scratch, never committed.
import { execFileSync } from "node:child_process";
import { readdirSync, readFileSync } from "node:fs";

const [phase, base, output] = process.argv.slice(2);
if (!/^6[2-7]$/.test(phase ?? "") || base === undefined || output === undefined) {
  console.error("usage: sweep-check.mjs <62..67> <phase_base> <reverify-output>");
  process.exit(2);
}
const dir = `.planning/phases/${readdirSync(".planning/phases").find((d) => d.startsWith(`QD-${phase}-`))}`;
const doc = `${dir}/${phase}-VERIFICATION.md`;
const before = execFileSync("git", ["show", `${base}:${doc}`], { encoding: "utf8" });
const now = readFileSync(doc, "utf8");
const lines = readFileSync(output, "utf8").trimEnd().split("\n");
const digest = lines.pop();
const block = lines.join("\n");

const DIGEST = /^covered_digest: .*$/m;
const drop = (text) => text.replace(DIGEST, "");
/** Data rows of the first table under the first heading matching `heading`, or -1 when there is no such heading. */
const rows = (text, heading) => {
  const at = text.search(heading);
  if (at === -1) return -1;
  const body = text.slice(at).split("\n").slice(1);
  const start = body.findIndex((line) => line.startsWith("|"));
  if (start === -1) return 0;
  let end = start;
  while (end < body.length && body[end].startsWith("|")) end++;
  return Math.max(0, end - start - 2);
};

const bad = [];
if (!/^covered_digest: /m.test(digest)) bad.push(`${output} does not end with a covered_digest line`);
if (DIGEST.exec(now)?.[0] !== digest) bad.push("the covered_digest line is not the recomputed one");
if (!drop(now).startsWith(drop(before))) bad.push("the report at the phase base is not kept whole before the new section");
const added = drop(now).slice(drop(before).length);
const SECTION = /^## Re-verification \(2026-11-21, Phase 68 sweep\)$/gm;
if ((now.match(SECTION) ?? []).length !== 1 || !added.includes("## Re-verification (2026-11-21, Phase 68 sweep)")) {
  bad.push("the appended section's heading is not there exactly once");
}
if (!added.includes(block)) bad.push("the appended section does not hold reverify.mjs's output verbatim");
const spot = rows(before, /^### Behaviou?ral Spot-Checks$/m);
const want = spot === -1 ? rows(before, /^### Observable Truths/m) : spot;
const got = rows(added, /^### Spot-checks re-run \(Phase 68 sweep\)$/m);
if (got < Math.max(want, 1)) bad.push(`the spot-checks re-run table has ${got} row(s), wanted at least ${Math.max(want, 1)}`);
if (phase === "62") {
  const truths = rows(before, /^### Observable Truths/m);
  const checked = rows(added, /^### Truths re-checked \(Phase 68 sweep\)$/m);
  if (checked < truths) bad.push(`the truths re-checked table has ${checked} row(s), wanted at least ${truths}`);
}
if (!/^\*\*Verdict:\*\* passed$/m.test(added)) bad.push("the section does not end in **Verdict:** passed");
if (bad.length > 0) {
  console.error(bad.join("\n"));
  process.exit(1);
}
console.log(`sweep-check: Phase ${phase} re-verified, ${got} spot-check row(s), verdict passed`);
````
