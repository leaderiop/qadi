````js
// node extract.mjs <plan.md> "<heading line>": prints the content of the first fenced block after that exact heading
// line, plus one final newline. The closing fence is the opening fence's backticks alone on a line. Scratch only.
import { readFileSync } from "node:fs";
const [file, heading] = process.argv.slice(2);
const lines = readFileSync(file, "utf8").split("\n");
const at = lines.indexOf(heading);
if (at < 0) throw new Error(`no line ${heading} in ${file}`);
const open = lines.findIndex((line, i) => i > at && /^`{3,}/.test(line));
if (open < 0) throw new Error(`no fenced block after ${heading}`);
const fence = lines[open].match(/^`+/)[0];
const close = lines.findIndex((line, i) => i > open && line === fence);
if (close < 0) throw new Error(`the block after ${heading} is not closed`);
process.stdout.write(`${lines.slice(open + 1, close).join("\n")}\n`);
````

````js
// node make-harness.mjs <gate.mjs> <out.mjs> <repo-root>: writes a harness that runs only the probe guard's parse block
// of <gate.mjs> (from the line `  let seen;` to the line before `  // The calls the run made`), with the gate's own
// `why` and `REFUSED_SHAPE_PROBES`. Run it as: node <out.mjs> <sandbox> <timeout-ms>; it prints {seen, failures} as JSON.
// Scratch, never committed.
import { readFileSync, writeFileSync } from "node:fs";

const [gate, out, root] = process.argv.slice(2);
const lines = readFileSync(gate, "utf8").split("\n");
const slice = (first, isLast) => {
  const from = lines.indexOf(first);
  if (from === -1) throw new Error(`${gate}: no line ${JSON.stringify(first)}`);
  const to = lines.findIndex((line, i) => i > from && isLast(line));
  if (to === -1) throw new Error(`${gate}: no end after ${JSON.stringify(first)}`);
  return lines.slice(from, to + 1);
};
const why = slice("const why = (error) =>", (line) => line.endsWith("|| error.message;"));
const probes = slice("const REFUSED_SHAPE_PROBES = new Map([", (line) => line === "]);");
const block = slice("  let seen;", (line) => line.startsWith("  // The calls the run made")).slice(0, -1);
writeFileSync(
  out,
  [
    'import { execFileSync } from "node:child_process";',
    'import { readFileSync } from "node:fs";',
    'import { join } from "node:path";',
    `const ROOT = ${JSON.stringify(root)};`,
    "const SANDBOX = process.argv[2];",
    "const CHILD_TIMEOUT_MS = Number(process.argv[3]);",
    "const failures = [];",
    "const fail = (message) => failures.push(message);",
    "void execFileSync;",
    ...why,
    ...probes,
    ...block,
    "console.log(JSON.stringify({ seen: seen && { typed: [...seen.typed].sort(), refused: [...seen.refused].sort() }, failures }));",
    "",
  ].join("\n"),
);
````
