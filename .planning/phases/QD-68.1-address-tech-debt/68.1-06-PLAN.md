// node reverify-check.mjs <62..67> <pre-ref> <reverify-output> <stale-output> [<label>], from the repository root: checks
// the re-verification section a 68.1 plan appended to that phase's VERIFICATION.md against the report at <pre-ref> (the
// commit before the edit), the output of `68-reverify.mjs` and the output of `68.1-stale.mjs`, both saved before the
// edit, under <label> (default the first round's). Prints `reverify-check: <NN> ok` or each problem on stderr and exits
// 1. node reverify-check.mjs --self-test plants one defect per check. Scratch, never committed.
import { execFileSync } from "node:child_process";
import { readdirSync, readFileSync } from "node:fs";

const LABEL = (process.argv[2] !== "--self-test" && process.argv[6]) || "Re-verification (2026-11-21, after Phase 68.1)";
const ORIGINAL = /^### (?:Behaviou?ral Spot-Checks|Observable Truths)\b/;

/** The rows of the first table after the first line matching `heading`: lines starting `|`, header and rule dropped. */
const rows = (lines, heading) => {
  const at = lines.findIndex((line) => heading.test(line));
  if (at === -1) return undefined;
  const start = lines.findIndex((line, i) => i > at && line.startsWith("|"));
  if (start === -1) return [];
  const end = lines.findIndex((line, i) => i > start && !line.startsWith("|"));
  return lines.slice(start + 2, end === -1 ? lines.length : end);
};

/** Every problem with `now` as a re-verification of `old` for the phase whose directory is `dir`; [] when none. */
export const problems = (phase, dir, old, now, reverify, stale) => {
  const out = [];
  const digest = (line) => line.startsWith("covered_digest:");
  const oldLines = old.replace(/\n+$/, "").split("\n");
  const nowLines = now.replace(/\n+$/, "").split("\n");
  const head = nowLines.slice(0, oldLines.length);
  if (head.filter((l) => !digest(l)).join("\n") !== oldLines.filter((l) => !digest(l)).join("\n"))
    out.push("the report before the section was edited, not only appended to");
  const printed = reverify.replace(/\n+$/, "").split("\n");
  const newDigest = printed.at(-1);
  if (nowLines.filter(digest).length !== 1 || nowLines.find(digest) !== newDigest)
    out.push("the covered_digest line is not the one 68-reverify.mjs printed last");
  const added = nowLines.slice(oldLines.length);
  const section = added.join("\n");
  if (added[0] !== "" || !section.startsWith(`\n${printed.slice(0, -1).join("\n")}\n`))
    out.push("the section does not open with 68-reverify.mjs's output, verbatim, after one blank line");
  if (printed[0] !== `## ${LABEL}` || added.filter((l) => l === `## ${LABEL}`).length !== 1)
    out.push(`the section heading is not "## ${LABEL}" exactly once`);
  if (printed.some((l) => l.includes("(before Phase 68)"))) out.push("a changed file is marked (before Phase 68)");
  const staleLines = stale.split("\n");
  const at = staleLines.findIndex((l) => l.startsWith(`${phase}: stale, `));
  const listed = [];
  for (let i = at + 1; at !== -1 && i < staleLines.length && staleLines[i].startsWith("  "); i++)
    listed.push(staleLines[i].trim().replace(/ \(covered\)$/, ""));
  if (at === -1 || !added.some((l) => l.startsWith(`**Staleness by \`68.1-stale.mjs\`:** \`${staleLines[at]}\``)))
    out.push("no line gives 68.1-stale.mjs's staleness line for the phase");
  const spot = rows(added, new RegExp(`^### Spot-checks re-run \\(${LABEL.replace(/[()]/g, "\\$&")}\\)$`));
  if (spot === undefined) out.push("no spot-checks re-run heading with the label");
  else if (spot.length === 0) out.push("the spot-checks re-run table has no row");
  else if (spot.some((r) => !/\|\s*PASS\s*\|\s*$/.test(r))) out.push("a spot-check row is not PASS");
  if (rows(oldLines, ORIGINAL) === undefined) out.push("the report has no spot-check or Observable Truths table");
  const fromTable = printed.filter((l) => /^\| `[^`]+`( \(covered\))? \| /.test(l)).map((l) => /^\| `([^`]+)`/.exec(l)[1]);
  const changed = [...new Set([...fromTable, ...listed])];
  const read = added.findIndex((l) => l === `### Changed cited files read against the truths that cite them (${LABEL})`);
  if (read === -1) out.push("no changed-cited-files heading with the label");
  else for (const f of changed) if (!added.slice(read).some((l) => l.startsWith(`- \`${f}\``))) out.push(`${f} is not read`);
  if (changed.length === 0) out.push("68-reverify.mjs listed no changed file: the phase was not stale");
  for (const m of section.matchAll(/\.planning\/[^\s`|)]*/g))
    if (!m[0].startsWith(`${dir}/`)) out.push(`the section names ${m[0]}, outside the phase's own directory`);
  if (nowLines.at(-1) !== "**Verdict:** passed") out.push("the last line is not **Verdict:** passed");
  return out;
};

const selfTest = () => {
  const dir = ".planning/phases/QD-62-x";
  const old = ["---", "covered_digest: old", "---", "### Behavioral Spot-Checks", "", "| B | C | R | S |", "| - | - | - | - |", "| a | b | c | PASS |", ""].join("\n");
  const reverify = [`## ${LABEL}`, "", "| Cited file | Changed since `x` by |", "| --- | --- |", "| `scripts/a.mjs` (covered) | `abc` fix |", "", "covered_digest: new"].join("\n");
  const section = [
    "",
    ...reverify.split("\n").slice(0, -1),
    "**Staleness by `68.1-stale.mjs`:** `62: stale, verification x, 3 cited, 1 changed since` (run at `y`).",
    "",
    `### Spot-checks re-run (${LABEL})`,
    "",
    "| B | C | R | S |",
    "| - | - | - | - |",
    "| a | b | c | PASS |",
    "",
    `### Changed cited files read against the truths that cite them (${LABEL})`,
    "",
    "- `scripts/a.mjs` (`abc`): still holds.",
    "",
    "**Verdict:** passed",
  ];
  const good = `${old.replace("covered_digest: old", "covered_digest: new").replace(/\n$/, "")}\n${section.join("\n")}\n`;
  const plants = [
    ["the report before the section was edited", good.replace("| a | b | c | PASS |\n", "| a | b | z | PASS |\n")],
    ["the covered_digest line is not", good.replace("covered_digest: new", "covered_digest: old")],
    ["does not open with 68-reverify.mjs's output", good.replace("| --- | --- |\n", "")],
    ["exactly once", good.replace("**Verdict:** passed", `## ${LABEL}\n**Verdict:** passed`)],
    ["no line gives 68.1-stale.mjs's staleness line", good.replace("**Staleness by", "**Staleness from")],
    ["no line gives 68.1-stale.mjs's staleness line", good.replace("1 changed since", "2 changed since")],
    ["no spot-checks re-run heading", good.replace(`### Spot-checks re-run (${LABEL})`, "### Spot-checks re-run")],
    ["a spot-check row is not PASS", good.replace(/\| a \| b \| c \| PASS \|\n\n### Changed/, "| a | b | c | FAIL |\n\n### Changed")],
    ["scripts/a.mjs is not read", good.replace("- `scripts/a.mjs` (`abc`)", "- `scripts/b.mjs` (`abc`)")],
    ["outside the phase's own directory", good.replace("still holds.", "still holds (.planning/REQUIREMENTS.md).")],
    ["the last line is not", good.replace("**Verdict:** passed", "**Verdict:** gaps_found")],
  ];
  const stale = "62: stale, verification x, 3 cited, 1 changed since\n  scripts/a.mjs (covered)\nredo: 62\n";
  const clean = problems("62", dir, old, good, reverify, stale);
  if (clean.length > 0) throw new Error(`self-test FAILED: the clean section: ${clean.join("; ")}`);
  for (const [want, now] of plants) {
    const got = problems("62", dir, old, now, reverify, stale);
    if (!got.some((p) => p.includes(want))) throw new Error(`self-test FAILED: planted "${want}": got ${JSON.stringify(got)}`);
  }
  const only = problems("62", dir, old, good, reverify, `${stale.replace("redo", "  scripts/c.mjs\nredo")}`);
  if (!only.some((p) => p.includes("scripts/c.mjs is not read")))
    throw new Error(`self-test FAILED: a file only 68.1-stale.mjs lists: got ${JSON.stringify(only)}`);
  console.log(`self-test: ${plants.length + 2} passed`);
};

if (process.argv[2] === "--self-test") selfTest();
else {
  const [phase, pre, output, staleOutput] = process.argv.slice(2);
  const dir = `.planning/phases/${readdirSync(".planning/phases").find((d) => d.startsWith(`QD-${phase}-`))}`;
  const doc = `${dir}/${phase}-VERIFICATION.md`;
  const old = execFileSync("git", ["show", `${pre}:${doc}`], { encoding: "utf8" });
  const found = problems(phase, dir, old, readFileSync(doc, "utf8"), readFileSync(output, "utf8"), readFileSync(staleOutput, "utf8"));
  for (const p of found) console.error(`reverify-check: ${phase}: ${p}`);
  if (found.length > 0) process.exit(1);
  console.log(`reverify-check: ${phase} ok`);
}
