```ts
// packages/devtools/src/model/GatesFeed.ts — package-private (ADR-QD-099), in neither barrel
import * as Data from "effect/Data";
import type * as Effect from "effect/Effect";
import type * as Scope from "effect/Scope";
import type { AskedQuestionLike, GateInstanceLike, GateRegistryLike } from "./Gates.ts";

export type RegistryRead = Data.TaggedEnum<{
  Unwatched: {};                                              // no registry handed
  Read: { readonly instances: ReadonlyArray<GateInstanceLike> }; // the registry's own array
  Failed: { readonly reason: string };                        // `instances` or `subscribe` threw
}>;
export const RegistryRead = Data.taggedEnum<RegistryRead>();

export type AskedRead = Data.TaggedEnum<{
  Unhanded: {};                                               // no thunk handed
  Read: { readonly questions: ReadonlyArray<AskedQuestionLike> };
  Failed: { readonly reason: string };                        // the thunk, or comparing its answer, threw
}>;
export const AskedRead = Data.taggedEnum<AskedRead>();

export interface GatesSnapshot {
  readonly registry: RegistryRead;
  readonly asked: AskedRead;
}

export interface GatesFeedInputs {
  readonly gates?: GateRegistryLike | undefined;
  readonly questions?: (() => ReadonlyArray<AskedQuestionLike>) | undefined;
  /** The thunk's cadence. Defaults to `DEFAULT_DIAGNOSTICS_INTERVAL_MILLIS`. */
  readonly intervalMillis?: number | undefined;
}

export interface GatesFeed {
  readonly subscribe: (listener: () => void) => () => void;
  readonly getSnapshot: () => GatesSnapshot;
  /** Re-reads the registry, re-subscribes when its identity moved or the last watch failed, reads a new thunk; publishes at most once. */
  readonly configure: (next: GatesFeedInputs) => void;
}

export declare const makeGatesFeed: (init?: GatesFeedInputs) => Effect.Effect<GatesFeed, never, Scope.Scope>;
```

## Flagged assumptions

- **The cadence starts one interval after build or configure** (research Q3): the build and each `configure` that
  hands a thunk read it themselves, so one `configure` reads each input at most once and the base's first-sample read is
  matched by the build's read.
- **The release finalizer is registered before `makeSupersedingRun()`**, so on close the run's finalizer (interrupt,
  `closed()` true) runs first and the listener the release removes can no longer reach host code.
- **A listener that throws inside a cadence publish** ends the cadence fiber, as one inside the poll's `accept` ends
  the poll. The feed's one listener in the dock is `DockModel`'s pure `rejoin`; recorded as an accepted threat.
- **T3's hostile answer** is a Proxy over an array whose `length` getter throws; if `Equal.equals` in the installed
  `effect` does not read `length` for two arrays, the executor picks the trap it does read (recorded in the SUMMARY),
  keeping the claim: a compare that throws is stated, not thrown.
