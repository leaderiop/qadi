### File: records-edit.mjs

Scratch, `<S>/records-edit.mjs`, also used by 68-10 and 68-11 (SHA-256 `1ddb1c8914f1acf73be7d799ed0f8ac8b2e14c1d95bd1dd0aba64ab820651c19`).

````js
// Applies one records edit list from the repository root: each edit replaces `old` with `new` where `old` occurs
// exactly once, appends `append`, or inserts `text` after the one line that starts with `after_line_starting`. An edit
// that does not find its text exactly once stops the run before any file is written. Scratch, never committed:
//   node <S>/records-edit.mjs <S>/texts/records-t1.json
import { readFileSync, writeFileSync } from "node:fs";

const [list] = process.argv.slice(2);
const edits = JSON.parse(readFileSync(list, "utf8"));
const files = new Map();
const stop = (message) => {
  console.error(message);
  process.exit(1);
};
for (const e of edits) {
  let s = files.get(e.file) ?? readFileSync(e.file, "utf8");
  if ("append" in e) s += e.append;
  else if ("after_line_starting" in e) {
    const lines = s.split("\n");
    const at = lines.flatMap((line, i) => (line.startsWith(e.after_line_starting) ? [i] : []));
    if (at.length !== 1) stop(`${e.file}: ${at.length} lines start with ${e.after_line_starting}`);
    lines.splice(at[0] + 1, 0, e.text.replace(/\n$/, ""));
    s = lines.join("\n");
  } else {
    const n = s.split(e.old).length - 1;
    if (n !== 1) stop(`${e.file}: the old text occurs ${n} times: ${e.old.slice(0, 80)}`);
    s = s.replace(e.old, () => e.new);
  }
  files.set(e.file, s);
}
for (const [file, s] of files) writeFileSync(file, s);
console.log(`${list}: ${edits.length} edit(s) applied to ${files.size} file(s)`);
````
