export interface DockModelInputs {
  readonly capacity?: number | undefined;
  readonly catalogue?: Catalogue | undefined;
  readonly ports?: EvaluationPortsLayer | undefined;
  /** Read in the model's own scope; a new source (or capacity) supersedes the previous read (D-14). */
  readonly source?: Source | undefined;
}
export interface DockModel {
  /** Stable across a capacity change (D-05). */
  readonly timeline: TimelineStore;
  readonly dock: DockSession;
  readonly explorer: ExplorerSession;
  readonly roles: HeldChoiceStore<Role>;
  readonly simulation: SimulationSession;
  /** catalogueOf(the shown timeline, the current catalogue), computed once per change. */
  readonly sightings: () => ReadonlyArray<PolicySighting>;
  /** Dispatches only the fields whose identity changed (D-03). */
  readonly configure: (inputs: DockModelInputs) => void;
  // [25] no `dispose`: the model, and the simulation session in it, end when the scope closes (D-13).
}
/** Effect.fn("qadi.devtools.makeDockModel"); builds the session in the same scope (D-13). */
export const makeDockModel: (init?: DockModelInputs) => Effect.Effect<DockModel, never, Scope.Scope>;
