```js
// The phase's close checks, each named, all run, and a non-zero exit if any fails. Scratch;
// never committed. From the repository root:
//   node <S>/close-scan.mjs <phase_base>
import { spawnSync } from "node:child_process";
import { existsSync, readFileSync } from "node:fs";

const BASE = process.argv[2];
const run = (cmd, args) =>
  spawnSync(cmd, args, { encoding: "utf8", stdio: ["ignore", "pipe", "pipe"], env: { ...process.env, NO_COLOR: "1" }, maxBuffer: 1 << 26 });
const git = (...args) => {
  const r = run("git", args);
  if (r.status !== 0) throw new Error(`git ${args.join(" ")} failed: ${r.stderr}`);
  return r.stdout;
};
/** Comments blanked and string contents blanked, length and newlines kept. */
const blank = (source) => {
  const out = source.split("");
  const wipe = (from, to) => {
    for (let k = from; k < to; k += 1) if (out[k] !== "\n") out[k] = " ";
  };
  let i = 0;
  while (i < source.length) {
    const pair = source.slice(i, i + 2);
    const c = source[i];
    if (pair === "//" || pair === "/*") {
      const end = pair === "//" ? source.indexOf("\n", i) : source.indexOf("*/", i + 2);
      const stop = end === -1 ? source.length : pair === "//" ? end : end + 2;
      wipe(i, stop);
      i = stop;
    } else if (c === '"' || c === "'" || c === "`") {
      let j = i + 1;
      while (j < source.length && source[j] !== c) j += source[j] === "\\" ? 2 : 1;
      wipe(i + 1, Math.min(j, source.length));
      i = j + 1;
    } else {
      i += 1;
    }
  }
  return out.join("");
};
const FORBIDDEN = {
  "type assertion": /\bas\b(?!\s+const\b)/g,
  "non-null assertion": /[\w)\]]!(?=[.[(;,)\s])/g,
  any: /\bany\b/g,
  switch: /\bswitch\s*\(/g,
  "Effect.fnUntraced": /\bfnUntraced\b/g,
  async: /\basync\b/g,
  await: /\bawait\b/g,
  "Record<string, unknown>": /Record<\s*string\s*,\s*unknown\s*>/g,
};
const TEST_ONLY = { console: /\bconsole\./g, "process.env": /\bprocess\.env\b/g, "Date.now": /\bDate\.now\b/g, "new Date": /\bnew Date\(/g };
const count = (text, patterns) =>
  Object.fromEntries(
    Object.entries(patterns).map(([name, re]) => [
      name,
      text
        .split("\n")
        .filter((l) => !/^\s*(import|export)\b/.test(l))
        .join("\n")
        .match(re)?.length ?? 0,
    ]),
  );
const results = [];
const check = (name, ok, detail) => {
  results.push({ name, ok, detail });
  console.log(`${ok ? "ok  " : "FAIL"} ${name}${detail ? `: ${detail}` : ""}`);
};

if (!/^[0-9a-f]{40}$/.test(BASE ?? "")) {
  console.error("usage: node close-scan.mjs <phase_base, 40 hex>");
  process.exit(2);
}

const src = git("diff", "--stat", BASE, "--", "packages/*/src").trim();
check("no packages/*/src change since phase_base", src === "", src);

const changesets = git("diff", "--name-only", "--diff-filter=A", BASE, "--", ".changeset").trim();
check("no changeset added", changesets === "", changesets);
const tags = git("tag", "--contains", BASE).trim();
check("no tag on the phase's commits", tags === "", tags);
const subjects = git("log", "--format=%s", `${BASE}..HEAD`).split("\n").filter((s) => /^(chore\(release\)|version packages|release\b)/i.test(s));
check("no release or version commit", subjects.length === 0, subjects.join("; "));

const touched = git("diff", "--name-only", BASE, "--", "*.mjs", "*.ts", "*.tsx", ":(exclude).planning")
  .split("\n")
  .filter((f) => f !== "" && existsSync(f));
const findings = [];
for (const file of touched) {
  const now = readFileSync(file, "utf8");
  const before = run("git", ["show", `${BASE}:${file}`]);
  const was = before.status === 0 ? before.stdout : "";
  const patterns = file.endsWith(".tst.ts") || file.includes("/test/") ? { ...FORBIDDEN, ...TEST_ONLY } : FORBIDDEN;
  const a = count(blank(was), patterns);
  const b = count(blank(now), patterns);
  for (const name of Object.keys(patterns)) if (b[name] > a[name]) findings.push(`${file}: ${name} ${a[name]} -> ${b[name]}`);
  const sd = (t) => t.split("Stryker disable").length - 1;
  if (sd(now) > sd(was)) findings.push(`${file}: Stryker disable ${sd(was)} -> ${sd(now)}`);
}
check(`no forbidden construct added (${touched.length} code file(s) scanned)`, findings.length === 0, findings.join("; "));

const added = git("diff", "-U0", BASE, "--", ".", ":(exclude).planning")
  .split("\n")
  .filter((l) => l.startsWith("+") && !l.startsWith("+++"));
const LEAK = /\bD-?\d{1,2}\b|\b(?:PIN|WX)-0\d\b|\b62-0\d\b|\b[Pp]hase 62\b|\bQD-62\b/;
const leaks = added.filter((l) => LEAK.test(l));
check(`no decision id, plan id or phase number added outside .planning (${added.length} added line(s))`, leaks.length === 0, leaks.slice(0, 5).join(" | "));

const hs = run(process.execPath, ["scripts/check-house-style.mjs"]);
check("house-style clean, 66 type-test files", hs.status === 0 && hs.stdout.includes("house-style: 560 file(s) clean (3 declared switch(es), 66 type-test file(s))"), hs.status === 0 ? "" : hs.stderr.slice(-400));
const agents = readFileSync("AGENTS.md", "utf8").split("\n").filter((l) => /^\| `packages\/[^`]+\.tst\.ts` \| \d+ \| \d+ \|/.test(l));
const totals = agents.reduce((t, l) => {
  const m = /\| (\d+) \| (\d+) \|/.exec(l);
  return m === null ? t : [t[0] + Number(m[1]), t[1] + Number(m[2])];
}, [0, 0]);
check("budget table 28 rows, 85 callable, 9 constructable", agents.length === 28 && totals[0] === 85 && totals[1] === 9, `${agents.length} rows, ${totals[0]}, ${totals[1]}`);

const api = run("pnpm", ["-s", "spec:api"]);
check("939 exports documented", api.status === 0 && `${api.stdout}${api.stderr}`.includes("api-surface: 939 export(s) documented across 9 package(s)"));
const dod = run(process.execPath, ["scripts/check-dod-table.mjs"]);
check("gate 11: 25 steps", dod.status === 0 && dod.stdout.includes("dod-table: 25 step(s) match pnpm check"));
const spec = run("bash", ["spec/scripts/verify-traceability.sh", "--strict"]);
check("verify-traceability --strict", spec.status === 0, spec.status === 0 ? "" : spec.stdout.split("\n").filter((l) => l.includes("FAIL")).join("; "));
const ts = run(process.execPath, ["scripts/run-tstyche.mjs"]);
const passes = `${ts.stdout}${ts.stderr}`.split("Test files: 66 passed, 66 total").length - 1;
check("both tstyche passes green on 66 files", ts.status === 0 && passes === 2, `exit ${ts.status}, ${passes} pass(es) of 66`);
check("no probe or flip copy left", !existsSync("packages/core/test/ZzRuleOnProbe.tst.ts") && git("ls-files", "--others", "--exclude-standard", "--", "packages").trim() === "");

const failed = results.filter((r) => !r.ok);
console.log(`close-scan: ${results.length - failed.length} of ${results.length} check(s) green`);
process.exit(failed.length === 0 ? 0 : 1);
```
