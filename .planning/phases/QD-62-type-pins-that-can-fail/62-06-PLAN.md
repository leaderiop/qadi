### `scripts/run-tstyche.mjs` (new)

```js
#!/usr/bin/env node
/**
 * Runs tstyche twice, so every type pin is checked under each setting of the Effect language
 * service's `floatingEffect` rule (ADR-QD-175).
 *
 * The first pass is tstyche's own configuration: `tstyche.json` names `tsconfig.tstyche.json`,
 * where the rule is off, so a negative `toBeCallableWith` pin on a function that returns an
 * `Effect` can fail. The second compiles under `tsconfig.test.json`, where the rule is on as it
 * is for `pnpm typecheck`; there a positive pin on such a function fails whatever its
 * arguments, so a bare negative pin on one cannot keep a passing positive twin. It skips the
 * one block that holds only with the rule off: the describe in `CHARACTERISATION` whose title
 * holds `RULE_ON_SKIP`. tstyche's `--skip` skips every describe or test whose title contains
 * the text, ignoring case, so the text must appear once in that file and in no other type test.
 *
 * Both passes always run, and the exit status is non-zero if either fails. No shell: the
 * arguments are fixed, and tstyche runs on the Node binary that runs this script.
 */
import { spawnSync } from "node:child_process";
import { readFileSync } from "node:fs";
import { join, relative } from "node:path";
import { fileURLToPath } from "node:url";

const ROOT = fileURLToPath(new URL("..", import.meta.url));
const TSTYCHE = join(ROOT, "node_modules", "tstyche", "dist", "bin.js");
const CHARACTERISATION = "packages/core/test/FloatingEffectPins.tst.ts";
const RULE_ON_SKIP = "under the tstyche tsconfig";

/**
 * @param {ReadonlyArray<string>} args
 * @param {"inherit" | "pipe"} stdout
 */
const tstyche = (args, stdout) =>
  spawnSync(process.execPath, [TSTYCHE, ...args], { cwd: ROOT, stdio: ["ignore", stdout, "inherit"], encoding: "utf8" });

const listed = tstyche(["--listFiles"], "pipe");
if (listed.status !== 0) {
  console.error("run-tstyche: `tstyche --listFiles` failed, so the skip text could not be checked.");
  process.exit(1);
}
/** @type {Array<{ readonly file: string, readonly count: number }>} */
const holders = JSON.parse(listed.stdout)
  .map((path) => {
    const file = relative(ROOT, path);
    return { file, count: readFileSync(join(ROOT, file), "utf8").toLowerCase().split(RULE_ON_SKIP).length - 1 };
  })
  .filter((holder) => holder.count > 0);
if (holders.length !== 1 || holders[0].file !== CHARACTERISATION || holders[0].count !== 1) {
  console.error(
    `run-tstyche: "${RULE_ON_SKIP}" must appear once, in ${CHARACTERISATION}, and in no other type test, ` +
      `because the second pass skips every title that contains it; found ${JSON.stringify(holders)}.`,
  );
  process.exit(1);
}

console.log("run-tstyche: pass 1 of 2, tstyche.json (tsconfig.tstyche.json, floatingEffect off)");
const first = tstyche([], "inherit");
console.log(`run-tstyche: pass 2 of 2, tsconfig.test.json (floatingEffect on), skipping "${RULE_ON_SKIP}"`);
const second = tstyche(["--tsconfig", "tsconfig.test.json", "--skip", RULE_ON_SKIP], "inherit");
process.exit(first.status === 0 && second.status === 0 ? 0 : 1);
```

```js
// RED for run-tstyche.mjs's guard on the skip text: each scenario edits one file, runs the
// script, writes the file back, and expects the guard to refuse before either pass. Scratch;
// never committed. From the repository root:
//   node <S>/red-guard.mjs
import { spawnSync } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";

const CHARACTERISATION = "packages/core/test/FloatingEffectPins.tst.ts";
const once = (s, a, b) => {
  if (s.split(a).length !== 2) throw new Error(`not exactly once: ${a}`);
  return s.replace(a, () => b);
};
const scenarios = [
  { name: "title renamed", file: CHARACTERISATION, edit: (s) => once(s, '"under the tstyche tsconfig, floatingEffect is off"', "\"under tstyche's own tsconfig, floatingEffect is off\""), fails: "found []" },
  { name: "planted in another type test", file: "packages/core/test/WalkEnvironment.tst.ts", edit: (s) => `${s}// under the tstyche tsconfig\n`, fails: '"file":"packages/core/test/WalkEnvironment.tst.ts"' },
  { name: "twice in the characterisation", file: CHARACTERISATION, edit: (s) => `${s}// Under The Tstyche Tsconfig\n`, fails: '"count":2' },
];
const bad = [];
for (const sc of scenarios) {
  const original = readFileSync(sc.file, "utf8");
  let r;
  try {
    writeFileSync(sc.file, sc.edit(original));
    r = spawnSync(process.execPath, ["scripts/run-tstyche.mjs"], { encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] });
  } finally {
    writeFileSync(sc.file, original);
  }
  const out = `${r.stdout}${r.stderr}`;
  const ok = r.status !== 0 && out.includes("must appear once") && out.includes(sc.fails) && !out.includes("pass 1 of 2");
  console.log(`${ok ? "ok " : "BAD"} ${sc.name}: exit ${r.status}`);
  if (!ok) bad.push(`${sc.name}: expected the guard to refuse with ${JSON.stringify(sc.fails)}\n${out.slice(-800)}`);
}
if (bad.length > 0) {
  console.error(bad.join("\n"));
  process.exit(1);
}
console.log(`guard RED: ${scenarios.length} scenario(s) refused`);
```
