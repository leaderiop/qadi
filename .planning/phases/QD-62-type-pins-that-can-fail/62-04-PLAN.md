 * tstyche's default `testFileMatch` (`**\/*.tst.*`, `**\/__typetests__/*.test.*`,

/**
 * `source` with every comment, and the inside of every string and template literal, replaced
 * by spaces. Newlines and length are kept, so an offset or a line number still points into the
 * original. With `keepStrings`, strings stay as written: that is the JSONC reading.
 * shortcut: a template literal's `${…}` is blanked with it and a regex literal is read as code; neither holds a pin in a type test, revisit if one does.
 *
 * @param {string} source
 * @param {{ readonly keepStrings?: boolean }} [options]
 * @returns {string}
 */
const blankCode = (source, { keepStrings = false } = {}) => {
  const out = source.split("");
  /** @param {number} from @param {number} to */
  const blank = (from, to) => {
    for (let k = from; k < to; k += 1) if (out[k] !== "\n") out[k] = " ";
  };
  let i = 0;
  while (i < source.length) {
    const pair = source.slice(i, i + 2);
    const c = source[i];
    if (pair === "//" || pair === "/*") {
      const end = pair === "//" ? source.indexOf("\n", i) : source.indexOf("*/", i + 2);
      const stop = end === -1 ? source.length : pair === "//" ? end : end + 2;
      blank(i, stop);
      i = stop;
    } else if (c === '"' || c === "'" || c === "`") {
      let j = i + 1;
      while (j < source.length && source[j] !== c) j += source[j] === "\\" ? 2 : 1;
      if (!keepStrings) blank(i + 1, Math.min(j, source.length));
      i = j + 1;
    } else {
      i += 1;
    }
  }
  return out.join("");
};

    const rel = relative(ROOT, full);
