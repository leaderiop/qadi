```js
// Reads a vitest JSON report: each "<title>=<count>" argument must fail exactly <count> times, nothing else may fail,
// every other test must pass, and no file may fail to load. With no title argument, every test must pass.
//   node red-check.mjs <report.json> ["<title>=<n>" ...]
import { readFileSync } from "node:fs";

const [file, ...wanted] = process.argv.slice(2);
const report = JSON.parse(readFileSync(file, "utf8"));
const results = report.testResults.flatMap((r) => r.assertionResults);
const want = new Map(
  wanted.map((w) => {
    const at = w.lastIndexOf("=");
    return [w.slice(0, at), Number(w.slice(at + 1))];
  }),
);
const failed = new Map();
for (const a of results.filter((r) => r.status === "failed")) failed.set(a.title, (failed.get(a.title) ?? 0) + 1);
const bad = [];
for (const [title, n] of want) {
  if ((failed.get(title) ?? 0) !== n) bad.push(`"${title}" failed ${failed.get(title) ?? 0} time(s), wanted ${n}`);
}
for (const [title, n] of failed) if (!want.has(title)) bad.push(`"${title}" failed ${n} time(s), wanted 0`);
const other = results.filter((r) => r.status !== "passed" && r.status !== "failed");
if (other.length > 0) bad.push(`${other.length} test(s) neither passed nor failed: ${other.map((r) => r.title).join(" | ")}`);
const unloaded = report.testResults.filter((r) => r.status === "failed" && r.assertionResults.length === 0);
if (unloaded.length > 0) bad.push(`a file failed to load: ${unloaded.map((r) => r.name).join(", ")}`);
if (results.length === 0) bad.push("the report holds no test");
if (bad.length > 0) {
  console.error(bad.join("\n"));
  process.exit(1);
}
const count = [...failed.values()].reduce((a, b) => a + b, 0);
console.log(`red-check: ${count} failed as named, ${results.filter((r) => r.status === "passed").length} passed`);
```
