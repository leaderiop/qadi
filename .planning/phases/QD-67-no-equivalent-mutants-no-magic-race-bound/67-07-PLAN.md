// node <S>/added-tokens.mjs <base>: the lines the working tree adds since <base> under packages/, searched for the
// tokens the house rules forbid, with `//` and `/* */` comment text and string and template contents blanked first,
// and, in the raw line, for a mutation-suppression comment and a milestone decision, plan, requirement or phase id (an
// ADR's own decision id, such as `ADR-QD-096 D-15-c`, is not one). The effect-discipline tokens (`async`, `await`,
// `console`, `process.env`, an ambient clock, id or promise) count in `packages/*/src` only, as house-style's
// non-test-scope rules do: a bench or test body may use them. Prints each hit as `<file>: <line>: <name>` and exits 1
// on any. Scratch only, never committed.
import { execFileSync } from "node:child_process";

const [base] = process.argv.slice(2);
const diff = execFileSync("git", ["diff", "-U0", base, "--", "packages"], { encoding: "utf8", maxBuffer: 1 << 26 });
const TOKENS = [
  [/(?<!\*\s*)\bas\s+[\w{[(<]/, "as"],
  [/\bany\b/, "any"],
  [/[\w)\]]!(?=[.;,)\]\s])/, "non-null !"],
  [/\bswitch\s*\(/, "switch"],
  [/\bfnUntraced\b/, "Effect.fnUntraced"],
  [/\basync\b/, "async"],
  [/\bawait\b/, "await"],
  [/\bconsole\./, "console"],
  [/\bprocess\.env\b/, "process.env"],
  [/\bDate\.now\b|\bnew Date\b|\bperformance\.now\b/, "ambient clock"],
  [/\bcrypto\.randomUUID\b|\bnew Promise\b|\.then\(/, "ambient id or promise"],
  [/Record<string,\s*unknown>/, "Record<string, unknown>"],
  [/Stryker disable/, "Stryker disable"],
  [/(?<!ADR-QD-\d{3} )\bD-?\d+[a-z-]*\b|\b(?:MUTE|WX)-\d+\b|\b67-0\d\b|\b[Pp]hase 67\b/, "milestone id"],
];
const RAW = new Set(["Stryker disable", "milestone id"]);
const SRC_ONLY = new Set(["async", "await", "console", "process.env", "ambient clock", "ambient id or promise"]);
const blank = (line) =>
  line
    .replace(/"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*'|`(?:[^`\\]|\\.)*`/g, '""')
    .replace(/\/\*.*?\*\//g, "")
    .replace(/\/\/.*$/, "")
    .replace(/^\s*\*.*$/, "")
    .replace(/^\s*\/\*\*?.*$/, "");
let file = "";
let hits = 0;
for (const raw of diff.split("\n")) {
  if (raw.startsWith("+++ b/")) file = raw.slice(6);
  if (!raw.startsWith("+") || raw.startsWith("+++")) continue;
  const code = raw.slice(1);
  // An import or export list renames with `as`; that is not an assertion.
  const stripped = /^\s*(?:import\b|export\s*(?:type\s*)?[{*])/.test(code) ? blank(code).replace(/\bas\s+\w+/g, "") : blank(code);
  for (const [re, name] of TOKENS) {
    if (SRC_ONLY.has(name) && !/^packages\/[^/]+\/src\//.test(file)) continue;
    const hit = RAW.has(name) ? re.test(code) : re.test(stripped);
    if (hit) {
      hits++;
      console.log(`${file}: ${code.trim()}: ${name}`);
    }
  }
}
console.log(`added-tokens: ${hits} hit(s)`);
process.exit(hits === 0 ? 0 : 1);
