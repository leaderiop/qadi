// Prints 59-BENCH.md's `## After-run` section from bench/after/ (meta.txt, the load files, analysis.tsv).
const fs = require("fs");
const cp = require("child_process");
const D = ".planning/phases/QD-59-admission-produces-the-one-walk-environment/";
const A = D + "bench/after/";
const meta = fs.readFileSync(A + "meta.txt", "utf8");
const v = (k) => (new RegExp("^" + k + ": (.*)$", "m").exec(meta) || [])[1];
const rows = fs.readFileSync(A + "analysis.tsv", "utf8").trimEnd().split("\n");
const h = Object.fromEntries(
  rows.filter((l) => l.startsWith("# ")).map((l) => {
    const s = l.slice(2);
    const i = s.indexOf(": ");
    return [s.slice(0, i), s.slice(i + 2)];
  }),
);
const n = (x) => Number(x).toLocaleString("en-US");
const signed = (x) => (Number(x) > 0 ? "+" : "") + x + "%";
const load = (f) => fs.readFileSync(A + f, "utf8").split("\n")[1].replace(/^.*load averages?: /, "");
const runs = fs.readdirSync(A).filter((f) => f.endsWith(".load-before")).sort().map((f) => f.replace(".load-before", ""));
const model = cp.execFileSync("sysctl", ["-n", "hw.model"], { encoding: "utf8" }).trim();
const out = [
  "",
  "## After-run",
  "",
  "The final source (HEAD `" + v("head") + "`) measured against the phase base with the same harness as the gate (D-06, D-18): " +
    "`bench/ab.sh bench/after/to-base.patch bench/after " + v("rounds") + " " + v("head_label") + " " + v("patched_label") + "`, " +
    "interleaved " + v("rounds") + " + " + v("rounds") + ", the base arm applying `to-base.patch` " +
    "(`git diff <head> <phase_base> -- packages/core/src`, SHA-256 `" + v("patch_sha256") + "`) and reversing it; " +
    "analysed by `bench/analyze.mjs bench/after " + v("patched_label") + " " + v("head_label") + "` " +
    "(reference base, candidate final). The statistic and the slower rule are the gate ones (D-04).",
  "",
  "Host: " + model + ", " + v("cpu") + "; Node " + v("node") + "; started " + v("start") + ".",
  "",
  "| run | load before (1, 5, 15 min) | load after |",
  "|---|---|---|",
  ...runs.map((r) => "| " + r + " | " + load(r + ".load-before") + " | " + load(r + ".load-after") + " |"),
  "",
  "Result: slower " + h.slower + " of " + h.workloads + "; geomean delta of minima " + h.geomean_d_min_pct + "%; worst " +
    h.worst_d_min + ". **gate: " + h.gate + "**",
  "",
  "Decision statistic: median of per-run minima (ns). Delta is final over base.",
  "",
  "| workload | base min (ns) | final min (ns) | delta | base spread | band | verdict |",
  "|---|---|---|---|---|---|---|",
];
for (const l of rows.slice(1)) {
  if (l.startsWith("#")) continue;
  const c = l.split("\t");
  out.push("| " + c[0] + " | " + n(c[1]) + " | " + n(c[2]) + " | " + signed(c[3]) + " | " + c[4] + "% | " + c[5] + "% | " + c[6] + " |");
}
process.stdout.write(out.join("\n") + "\n");
